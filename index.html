const operators = ['+', '-', '*', '/', '×', '÷']; function handleInputOperator(currentFormula, nextOperator) { if (!currentFormula) return ''; const lastChar = currentFormula.slice(-1); // 直前が演算子の場合は置き換え if (operators.includes(lastChar)) { return currentFormula.slice(0, -1) + nextOperator; } return currentFormula + nextOperator; }
